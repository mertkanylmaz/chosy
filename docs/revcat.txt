Adım 4: Tier Detection Logic'i — Hangi User Hangi Plan?
Entitlement aynı ama plan'ı nasıl ayırt edeceksin (free quota vs premium quota)?
RevenueCat CustomerInfo objesinde entitlements.active['chosy_plus'].productIdentifier field'ı var. Bu kullanıcının şu an aktif olan product ID'sini verir.
Frontend logic:
typescript// services/subscriptionService.ts
import Purchases from 'react-native-purchases'

export async function getUserTier(): Promise<SubscriptionTier> {
  const customerInfo = await Purchases.getCustomerInfo()
  const entitlement = customerInfo.entitlements.active['chosy_plus']
  
  if (!entitlement) {
    return 'free'
  }
  
  const productId = entitlement.productIdentifier
  
  // Map product ID to tier
  switch (productId) {
    case 'com.chosy.plus.lifetime.founding':
      return 'lifetime'
    
    case 'com.chosy.plus.annual.v2':
    case 'com.chosy.yearly': // legacy
      return 'annual'
    
    case 'com.chosy.plus.monthly.v2':
    case 'com.chosy.monthly': // legacy
      return 'monthly'
    
    case 'com.chosy.weekly': // legacy weekly user
      return 'weekly_legacy'
    
    default:
      // Unknown product → conservative, give monthly access
      console.warn('Unknown product ID:', productId)
      return 'monthly'
  }
}

8. SUPABASE WEBHOOK — Server-Side Truth
Client-side asla güvenilir değil. RevenueCat webhook'unu Supabase Edge Function'a bağla.
Webhook Setup
RevenueCat Dashboard → Project → Integrations → Webhooks → Add Webhook
URL: https://[your-project].supabase.co/functions/v1/revenuecat-webhook
Authorization Header Value: [random secret, RC_WEBHOOK_SECRET]
Edge Function
supabase/functions/revenuecat-webhook/index.ts:
typescriptimport { createClient } from '@supabase/supabase-js'

const RC_WEBHOOK_SECRET = Deno.env.get('RC_WEBHOOK_SECRET')!

export default async (req: Request) => {
  // 1. Verify secret
  const auth = req.headers.get('authorization')
  if (auth !== `Bearer ${RC_WEBHOOK_SECRET}`) {
    return new Response('Unauthorized', { status: 401 })
  }
  
  const event = await req.json()
  const { type, app_user_id, product_id } = event.event
  
  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  )
  
  // 2. Map product_id to tier
  const tier = mapProductToTier(product_id)
  
  // 3. Handle event types
  switch (type) {
    case 'INITIAL_PURCHASE':
    case 'RENEWAL':
    case 'PRODUCT_CHANGE':
      await supabase
        .from('users')
        .update({ 
          subscription_tier: tier,
          subscription_active_until: event.event.expiration_at_ms 
            ? new Date(event.event.expiration_at_ms).toISOString()
            : null, // null = lifetime
          updated_at: new Date().toISOString()
        })
        .eq('id', app_user_id)
      
      // Lifetime ise: claim_lifetime_spot RPC çağır
      if (tier === 'lifetime') {
        await supabase.rpc('claim_lifetime_spot', {
          p_user_id: app_user_id,
          p_price: event.event.price_in_purchased_currency
        })
      }
      break
    
    case 'CANCELLATION':
    case 'EXPIRATION':
      // Expire'a kadar erişim devam, sadece flag set
      await supabase
        .from('users')
        .update({ 
          subscription_will_renew: false,
          updated_at: new Date().toISOString()
        })
        .eq('id', app_user_id)
      
      // Win-back sequence trigger
      await supabase.from('winback_queue').insert({
        user_id: app_user_id,
        churned_at: new Date().toISOString()
      })
      break
    
    case 'BILLING_ISSUE':
      // User'a notification + retry logic
      await supabase.from('notification_log').insert({
        user_id: app_user_id,
        type: 'billing_issue',
        title: 'Ödeme sorunu',
        body: 'Aboneliğin yenilenirken sorun oluştu. Apple ID ödeme bilgini güncelle.'
      })
      break
  }
  
  return new Response('OK', { status: 200 })
}

function mapProductToTier(productId: string): string {
  if (productId === 'com.chosy.plus.lifetime.founding') return 'lifetime'
  if (productId.includes('annual') || productId.includes('yearly')) return 'annual'
  if (productId === 'com.chosy.weekly') return 'weekly_legacy'
  if (productId.includes('monthly')) return 'monthly'
  return 'free'
}
