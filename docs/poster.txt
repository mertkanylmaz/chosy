// Cron: her gün 23:00 UTC → ertesi günün puzzle'ı

import { createClient } from 'jsr:@supabase/supabase-js@2'

interface CurationCriteria {
  min_popularity: number
  min_vote_count: number
  exclude_recent_days: number
  difficulty_distribution: { easy: number, medium: number, hard: number }
}

const WEEKLY_DIFFICULTY_PATTERN = [
  'easy',    // Pazartesi
  'medium',  // Salı
  'easy',    // Çarşamba
  'medium',  // Perşembe
  'hard',    // Cuma
  'medium',  // Cumartesi
  'easy'     // Pazar
] as const

Deno.serve(async (req) => {
  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  )

  const tomorrow = new Date()
  tomorrow.setUTCDate(tomorrow.getUTCDate() + 1)
  const tomorrowDate = tomorrow.toISOString().split('T')[0]
  const dayOfWeek = tomorrow.getUTCDay()

  const { data: existing } = await supabase
    .from('daily_puzzles')
    .select('id')
    .eq('puzzle_date', tomorrowDate)
    .maybeSingle()

  if (existing) {
    return new Response(JSON.stringify({ status: 'already_curated' }))
  }

  const targetDifficulty = WEEKLY_DIFFICULTY_PATTERN[dayOfWeek]

  const popularityBounds = {
    easy: { min: 50, max: 100 },
    medium: { min: 25, max: 70 },
    hard: { min: 8, max: 35 }
  }[targetDifficulty]

  const usedRecentlyDate = new Date()
  usedRecentlyDate.setDate(usedRecentlyDate.getDate() - 180)

  const { data: candidates, error } = await supabase
    .rpc('get_puzzle_candidates', {
      p_popularity_min: popularityBounds.min,
      p_popularity_max: popularityBounds.max,
      p_min_vote_count: 500,
      p_exclude_after_date: usedRecentlyDate.toISOString().split('T')[0],
      p_limit: 20
    })

  if (error || !candidates?.length) {
    console.error('CURATION_FAILED', error)
    return new Response('Curation failed', { status: 500 })
  }

  const validated = await validatePosterQuality(candidates)
  const chosen = validated[Math.floor(Math.random() * Math.min(5, validated.length))]

  await supabase.from('daily_puzzles').insert({
    puzzle_date: tomorrowDate,
    film_id: chosen.id,
    difficulty_tier: targetDifficulty,
    curator_notes: `Auto: popularity ${chosen.popularity}, votes ${chosen.vote_count}`
  })

  return new Response(JSON.stringify({ 
    status: 'curated', 
    date: tomorrowDate,
    film_id: chosen.id,
    difficulty: targetDifficulty
  }))
})

async function validatePosterQuality(candidates: any[]) {
  return candidates.filter(c => 
    c.poster_url && 
    c.poster_url.includes('/w500/') &&
    c.original_language !== null
  )
}
Postgres helper RPC:
sqlCREATE OR REPLACE FUNCTION get_puzzle_candidates(
  p_popularity_min FLOAT,
  p_popularity_max FLOAT,
  p_min_vote_count INT,
  p_exclude_after_date DATE,
  p_limit INT DEFAULT 20
) RETURNS TABLE (
  id BIGINT,
  title TEXT,
  poster_url TEXT,
  popularity FLOAT,
  vote_count INT,
  original_language TEXT
) AS $$
  SELECT 
    f.id, f.title, f.poster_url, f.popularity, 
    f.vote_count, f.original_language
  FROM films f
  WHERE f.popularity BETWEEN p_popularity_min AND p_popularity_max
    AND f.vote_count >= p_min_vote_count
    AND f.poster_url IS NOT NULL
    AND f.id NOT IN (
      SELECT film_id FROM daily_puzzles 
      WHERE puzzle_date > p_exclude_after_date
    )
  ORDER BY RANDOM()
  LIMIT p_limit;
$$ LANGUAGE SQL STABLE;
Edge Function: Guess Validator
typescript// supabase/functions/submit-puzzle-guess/index.ts

interface GuessRequest {
  puzzle_date: string
  guess_film_id?: number
  guess_text?: string
}

Deno.serve(async (req) => {
  const authHeader = req.headers.get('Authorization')
  if (!authHeader) return new Response('Unauthorized', { status: 401 })

  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!,
    { global: { headers: { Authorization: authHeader } } }
  )

  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return new Response('Unauthorized', { status: 401 })

  const { puzzle_date, guess_film_id, guess_text }: GuessRequest = await req.json()

  const { data: puzzle } = await supabase
    .from('daily_puzzles')
    .select('id, film_id, films(title, original_title, alternative_titles)')
    .eq('puzzle_date', puzzle_date)
    .single()

  if (!puzzle) return new Response('Puzzle not found', { status: 404 })

  let attempt = await getOrCreateAttempt(supabase, user.id, puzzle.id)

  if (attempt.result !== 'in_progress') {
    return new Response(JSON.stringify({ error: 'ALREADY_COMPLETED', attempt }), { status: 400 })
  }

  if (attempt.attempts_used >= 6) {
    return new Response(JSON.stringify({ error: 'NO_ATTEMPTS_LEFT' }), { status: 400 })
  }

  const isCorrect = guess_film_id 
    ? guess_film_id === puzzle.film_id
    : await fuzzyMatchTitle(guess_text!, puzzle.films)

  const newAttemptsUsed = attempt.attempts_used + 1
  const newGuesses = [...attempt.guesses, {
    guess: guess_text || `film_id:${guess_film_id}`,
    correct: isCorrect,
    timestamp: new Date().toISOString()
  }]

  let result: 'in_progress' | 'won' | 'lost' = 'in_progress'
  let wonOnAttempt = null

  if (isCorrect) {
    result = 'won'
    wonOnAttempt = newAttemptsUsed
  } else if (newAttemptsUsed >= 6) {
    result = 'lost'
  }

  const { data: updated } = await supabase
    .from('puzzle_attempts')
    .update({
      attempts_used: newAttemptsUsed,
      guesses: newGuesses,
      result,
      won_on_attempt: wonOnAttempt,
      completed_at: result !== 'in_progress' ? new Date().toISOString() : null
    })
    .eq('id', attempt.id)
    .select()
    .single()

  if (result !== 'in_progress') {
    await updateStreak(supabase, user.id, result === 'won', puzzle_date)
  }

  const responsePayload: any = {
    correct: isCorrect,
    attempts_used: newAttemptsUsed,
    attempts_remaining: 6 - newAttemptsUsed,
    result,
    pixelation_level: getPixelationLevel(newAttemptsUsed)
  }

  if (!isCorrect && result === 'in_progress') {
    responsePayload.hint = await getHintForAttempt(supabase, puzzle.film_id, newAttemptsUsed, user.id)
  }

  if (result !== 'in_progress') {
    responsePayload.film = await getFullFilmDetails(supabase, puzzle.film_id)
  }

  return new Response(JSON.stringify(responsePayload), {
    headers: { 'Content-Type': 'application/json' }
  })
})

async function fuzzyMatchTitle(guess: string, film: any): Promise<boolean> {
  const normalize = (s: string) => 
    s.toLowerCase()
     .normalize('NFD')
     .replace(/[\u0300-\u036f]/g, '')
     .replace(/[^a-z0-9]/g, '')

  const normalizedGuess = normalize(guess)
  const titles = [film.title, film.original_title, ...(film.alternative_titles || [])]
  
  return titles.some(t => normalize(t) === normalizedGuess)
}

function getPixelationLevel(attemptsUsed: number): number {
  const levels = [60, 40, 24, 14, 8, 4]
  return levels[Math.min(attemptsUsed, 5)]
}

async function getHintForAttempt(
  supabase: any, 
  filmId: number, 
  attemptNumber: number,
  userId: string
): Promise<string> {
  const { data: user } = await supabase
    .from('users')
    .select('archetype')
    .eq('id', userId)
    .single()

  const { data: film } = await supabase
    .from('films')
    .select('genres, release_year, director, cast, mood_tags')
    .eq('id', filmId)
    .single()

  const archetype = user?.archetype || 'default'
  
  const hintMap: Record<string, string[]> = {
    default: [
      `Tür: ${film.genres[0]}`,
      `Çıkış: ${Math.floor(film.release_year / 10) * 10}'ler`,
      `Yönetmen: ${film.director}`,
      `Cast: ${film.cast?.[0]}`,
      `Atmosfer: ${film.mood_tags?.[0]}`
    ],
    'The Analyst': [
      `Tür: ${film.genres.join(' / ')}`,
      `Yıl: ${film.release_year}`,
      `Yönetmen: ${film.director}`,
      `Senaryo: ${film.writer || film.director}`,
      `IMDB rating: ${film.vote_average}`
    ],
    'The Dreamer': [
      `Bir his: ${film.mood_tags?.[0]}`,
      `Dönem: ${Math.floor(film.release_year / 10) * 10}'ler`,
      `Atmosfer: ${film.mood_tags?.[1]}`,
      `Ana karakter: ${film.cast?.[0]}`,
      `Tür: ${film.genres[0]}`
    ]
  }

  const hints = hintMap[archetype] || hintMap.default
  return hints[attemptNumber - 1] || hints[hints.length - 1]
}

async function updateStreak(
  supabase: any,
  userId: string,
  won: boolean,
  puzzleDate: string
) {
  const { data: streak } = await supabase
    .from('puzzle_streaks')
    .select('*')
    .eq('user_id', userId)
    .maybeSingle()

  const today = new Date(puzzleDate)
  const yesterday = new Date(today)
  yesterday.setDate(yesterday.getDate() - 1)
  const yesterdayStr = yesterday.toISOString().split('T')[0]

  const continuingStreak = streak?.last_played_date === yesterdayStr

  const newCurrentStreak = won 
    ? (continuingStreak ? streak.current_streak + 1 : 1)
    : 0

  const newLongestStreak = Math.max(streak?.longest_streak || 0, newCurrentStreak)

  await supabase.from('puzzle_streaks').upsert({
    user_id: userId,
    current_streak: newCurrentStreak,
    longest_streak: newLongestStreak,
    last_played_date: puzzleDate,
    total_wins: (streak?.total_wins || 0) + (won ? 1 : 0),
    total_attempts: (streak?.total_attempts || 0) + 1,
    updated_at: new Date().toISOString()
  })
}

import { Canvas, Image, useImage, Mask, Rect, ImageShader } from '@shopify/react-native-skia'
import { useEffect, useState } from 'react'
import { View, Text, TextInput, Pressable } from 'react-native'
import { supabase } from '@/lib/supabase'

const PIXELATION_LEVELS = [60, 40, 24, 14, 8, 4]

export default function PosterleScreen() {
  const [puzzle, setPuzzle] = useState<DailyPuzzle | null>(null)
  const [attemptsUsed, setAttemptsUsed] = useState(0)
  const [hints, setHints] = useState<string[]>([])
  const [gameState, setGameState] = useState<'in_progress' | 'won' | 'lost'>('in_progress')
  const [revealedFilm, setRevealedFilm] = useState<Film | null>(null)
  
  const poster = useImage(puzzle?.poster_url ?? null)
  const pixelSize = PIXELATION_LEVELS[Math.min(attemptsUsed, 5)]

  useEffect(() => {
    loadTodaysPuzzle()
  }, [])

  const loadTodaysPuzzle = async () => {
    const today = new Date().toISOString().split('T')[0]
    
    const { data: puzzleData } = await supabase
      .from('daily_puzzles')
      .select('id, puzzle_date, films(poster_url)')
      .eq('puzzle_date', today)
      .single()

    const { data: attemptData } = await supabase
      .from('puzzle_attempts')
      .select('*')
      .eq('puzzle_id', puzzleData?.id)
      .maybeSingle()

    if (attemptData) {
      setAttemptsUsed(attemptData.attempts_used)
      setGameState(attemptData.result)
      setHints(extractHintsFromGuesses(attemptData.guesses))
    }

    setPuzzle({ ...puzzleData, poster_url: puzzleData.films.poster_url })
  }

  const submitGuess = async (filmId: number) => {
    const { data, error } = await supabase.functions.invoke('submit-puzzle-guess', {
      body: {
        puzzle_date: puzzle?.puzzle_date,
        guess_film_id: filmId
      }
    })

    if (error) {
      Toast.show({ type: 'error', text1: 'Bağlantı sorunu' })
      return
    }

    setAttemptsUsed(data.attempts_used)
    
    if (data.hint) {
      setHints([...hints, data.hint])
    }

    if (data.result !== 'in_progress') {
      setGameState(data.result)
      setRevealedFilm(data.film)
      Haptics.notificationAsync(
        data.result === 'won' 
          ? Haptics.NotificationFeedbackType.Success
          : Haptics.NotificationFeedbackType.Warning
      )
    } else {
      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium)
    }
  }

  if (!puzzle || !poster) return <LoadingState />

  return (
    <View style={{ flex: 1, padding: 20 }}>
      <Canvas style={{ width: '100%', aspectRatio: 2/3 }}>
        {gameState === 'in_progress' ? (
          <PixelatedPoster image={poster} pixelSize={pixelSize} />
        ) : (
          <Image image={poster} fit="contain" x={0} y={0} width={POSTER_WIDTH} height={POSTER_HEIGHT} />
        )}
      </Canvas>

      <AttemptIndicator current={attemptsUsed} max={6} />
      
      <HintsList hints={hints} />

      {gameState === 'in_progress' && (
        <GuessInput 
          onSubmit={submitGuess}
          poolFilter={{ 
            year_range: extractYearHint(hints),
            genre: extractGenreHint(hints) 
          }}
        />
      )}

      {gameState === 'won' && (
        <WinScreen 
          film={revealedFilm} 
          attemptsUsed={attemptsUsed}
          onShare={() => generateShareCard('win', attemptsUsed)}
        />
      )}

      {gameState === 'lost' && (
        <LossDiscoveryScreen 
          film={revealedFilm}
          onAddToWatchlist={() => addToWatchlist(revealedFilm!.id)}
          onExploreMood={() => router.push(`/mood?seed=${revealedFilm!.id}`)}
        />
      )}
    </View>
  )
}

function PixelatedPoster({ image, pixelSize }: { image: any, pixelSize: number }) {
  const POSTER_WIDTH = 300
  const POSTER_HEIGHT = 450
  const cols = Math.ceil(POSTER_WIDTH / pixelSize)
  const rows = Math.ceil(POSTER_HEIGHT / pixelSize)

  return (
    <Mask
      mode="luminance"
      mask={
        <Rect x={0} y={0} width={POSTER_WIDTH} height={POSTER_HEIGHT} color="white" />
      }
    >
      <Image 
        image={image} 
        x={0} 
        y={0} 
        width={POSTER_WIDTH} 
        height={POSTER_HEIGHT}
        sampling={{ filter: 'nearest' }}
      />
    </Mask>
  )
}

          {result === 'win' ? `${attemptsUsed}/6` : 'X/6'}
        </Text>
        <View style={shareStyles.grid}>
          {[0,1,2,3,4,5].map(i => (
            <Text key={i}>
              {i < attemptsUsed - 1 ? '🎬' : i === attemptsUsed - 1 && result === 'win' ? '✅' : '⬛'}
            </Text>
          ))}
        </View>
        <Text style={shareStyles.archetype}>{archetype}</Text>
        {streak > 2 && <Text style={shareStyles.streak}>🔥 {streak} day streak</Text>}
        <Text style={shareStyles.brand}>chosy.ai</Text>
      </View>
    </ViewShot>
  )
}

🚨 CTO PROAKTİF UYARILAR
1. App Store review riski:
Apple's Wordle-clone takedown'ları kanlıdır. "Posterle" ismi marka çatışması riski taşır. Önerim: "Daily Reel" veya "Frame It" — sinema metaforu, Wordle echo yok.
2. TMDb telif zonu:
Posterleri modifying (pixelation) TMDb terms'in cannot modify maddesine girer mi? Cevap muğlak. Çözüm: TMDb API attribution + "Powered by TMDb" footer mecbur. Yine de risk var — ücretsiz feature olarak tutun, paywall arkasına koymayın (commercial use argümanı zayıflar).
3. Aynı film tekrarı:
180 günlük exclusion penceresi koydum ama TMDb'de mood-aware filtreleme yapıldıktan sonra havuz daralıyor. Hedef: 500+ uygun film. Daha az olursa repeat'ler hızlanır → kullanıcı şikayeti. Curation algorithm'ini monitör et.
4. Cron singleton problem:
Supabase cron'lar at-least-once delivery garanti eder. daily_puzzles.puzzle_date UNIQUE constraint double-execution'a karşı koruyor ama curation race condition olabilir — function'a advisory lock ekle:
sqlSELECT pg_advisory_xact_lock(hashtext('curate_daily_puzzle'));
5. Streak grief mekaniği:
Streak kaybetmenin acısı çok keskin. Wordle bunu çözdü mü? Kısmen. "Streak freeze" mekaniği ekle — premium kullanıcılar ayda 1 kez streak donduralabilir (vacation mode). Bu hem retention hem premium dönüşüm tetikleyicisi.
